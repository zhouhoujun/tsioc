  3. 在 AgentRuntime.ts 新增 helper，核心只做三件事：
  - 只保留 enabled === true 的 bundles
  - 给每个 bundle 算出 activeTools / inactiveTools
  - 给 prompt 一个统一 activation hint

  建议返回形状：
  interface ToolPromptContext {
    bundles: Array<{
      name: string;
      description?: string;
      deferredActivation: boolean;
      activeTools: string[];
      inactiveTools: string[];
    }>;
    activation: {
      discoveryTools: string[];
      hasInactiveTools: boolean;
    };
  }

  实现可以很直接：

  private async buildToolPromptContext(
    sessionId: string,
    tools: AgentToolDefinition[]
  ): Promise<ToolPromptContext> {
    const visibleToolNames = new Set(tools.map(t => t.name));
    const enabledBundles = this.bundles.filter(bundle => bundle.enabled);

    const bundles = await Promise.all(enabledBundles.map(async bundle => {
      const bundleTools = bundle.tools.filter(name => visibleToolNames.has(name));
      const states = await Promise.all(bundleTools.map(async name => ({
        name,
        active: await this.toolRegistry.isToolActive(sessionId, name)
      })));

      return {
        name: bundle.name,
        deferredActivation: !!bundle.deferredActivation,
        activeTools: states.filter(s => s.active).map(s => s.name),
        inactiveTools: states.filter(s => !s.active).map(s => s.name)
      };
    }));
    
    return {
      bundles: bundles.filter(bundle => bundle.activeTools.length || bundle.inactiveTools.length),
      activation: {
        discoveryTools: ['tool_search', 'tool_inspect'],
        hasInactiveTools: bundles.some(bundle => bundle.inactiveTools.length > 0)
      } 
    };
  } 

  4. packages/agents/agent/src/prompt/sections/ToolsSection.ts 不要新建 section，直接增强现有 section。
  现在它只有平铺工具列表；这里最适合把 activation 状态直接标到工具名上，再追加一个简短的 bundle 摘要。这样模型不用自己推断。
  
  建议输出逻辑改成：
  
  - ## Available Tools
  - 每个 tool 行带状态：
    - - read_file [active]: ...
    - - glob_search [inactive]: ...
  - 如果 hasInactiveTools 为 true，加一句固定提示：
    - Inactive tools require discovery/activation through tool_search or tool_inspect before calling them.
  - 然后输出 ## Capability Bundles
    - - filesystem: active [read_file], inactive [glob_search, content_search]
    - - registry: active [tool_search, tool_inspect]
  
  可以用 context.extra?.toolContext 做，不需要改 prompt builder 公共接口。
  
  一个够小的实现方式：

  const toolContext = context.extra?.toolContext;
  const activeNames = new Set(
    toolContext?.bundles?.flatMap((bundle: any) => bundle.activeTools) ?? []
  );
  const inactiveNames = new Set(
    toolContext?.bundles?.flatMap((bundle: any) => bundle.inactiveTools) ?? []
  );

  const lines: string[] = ['## Available Tools'];
  for (const tool of tools) {
    const state = activeNames.has(tool.name)
      ? 'active'
      : inactiveNames.has(tool.name)
        ? 'inactive'
        : 'available';
    lines.push(`- ${tool.name} [${state}]: ${tool.description}`);
  }

  if (toolContext?.activation?.hasInactiveTools) {
    lines.push('');
    lines.push('Inactive tools require discovery or inspection through tool_search or tool_inspect before use.');
  }

    lines.push('');
    lines.push('## Capability Bundles');
    for (const bundle of toolContext.bundles) {
      const parts: string[] = [];
      if (bundle.activeTools.length) {
        parts.push(`active [${bundle.activeTools.join(', ')}]`);
      }
      if (bundle.inactiveTools.length) {
        parts.push(`inactive [${bundle.inactiveTools.join(', ')}]`);
      }
      const suffix = parts.length ? `: ${parts.join('; ')}` : '';
      lines.push(`- ${bundle.name}${bundle.deferredActivation ? ' (deferred)' : ''}${suffix}`);
    }
  }
